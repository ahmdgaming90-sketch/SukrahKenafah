const StoreProvider = ({ children }) => {
  const [users, setUsers] = useState(INITIAL_USERS);
  const [products, setProducts] = useState(SEED_PRODUCTS);
  const [categories] = useState(SEED_CATEGORIES);
  const [settings, setSettings] = useState(INITIAL_SETTINGS);
  const [activeTheme, setActiveTheme] = useState('light');
  
  const [currentUser, setCurrentUser] = useState(null); // Auth State
  const [cart, setCart] = useState([]);
  const [isCartOpen, setIsCartOpen] = useState(false);
  const [selectedProduct, setSelectedProduct] = useState(null);
  const [isSearchOpen, setIsSearchOpen] = useState(false);

  // --- Dynamic Theme Injection ---
  useEffect(() => {
    if (!document.getElementById('alexandria-font')) {
      const link = document.createElement('link');
      link.id = 'alexandria-font';
      link.href = 'https://fonts.googleapis.com/css2?family=Alexandria:wght@300;400;600;800&display=swap';
      link.rel = 'stylesheet';
      document.head.appendChild(link);
    }
    
    const theme = THEMES[activeTheme].colors;
    const style = document.createElement('style');
    style.innerHTML = `
      :root {
        --color-primary: ${theme.primary};
        --color-secondary: ${theme.secondary};
        --color-bg: ${theme.bg};
        --color-card-bg: ${theme.cardBg};
        --color-text: ${theme.text};
        --color-text-muted: ${theme.textMuted};
      }
      body { 
        font-family: 'Alexandria', sans-serif; 
        background-color: var(--color-bg);
        color: var(--color-text);
      }
      .theme-bg { background-color: var(--color-bg); }
      .theme-card { background-color: var(--color-card-bg); box-shadow: 0 10px 40px -10px rgba(0,0,0,0.08); border: 1px solid rgba(150,150,150,0.1); }
      .theme-text { color: var(--color-text); }
      .theme-text-muted { color: var(--color-text-muted); }
      .theme-primary-bg { background-color: var(--color-primary); color: #fff; }
      .theme-primary-text { color: var(--color-primary); }
      .glass-nav { background: color-mix(in srgb, var(--color-bg) 85%, transparent); backdrop-filter: blur(16px); -webkit-backdrop-filter: blur(16px); border-bottom: 1px solid rgba(150,150,150,0.1); }
    `;
    const oldStyle = document.getElementById('theme-styles');
    if (oldStyle) oldStyle.remove();
    style.id = 'theme-styles';
    document.head.appendChild(style);
  }, [activeTheme]);

  // --- Cart Logic ---
  const addToCart = (item) => {
    const itemId = `${item.product.id}-${item.variant?.id || 'none'}-${item.addons.map(a=>a.id).sort().join('_')}-${item.notes}`;
    const exist = cart.findIndex(c => c.cartId === itemId);
    if (exist > -1) {
      const newCart = [...cart]; newCart[exist].qty += item.qty; setCart(newCart);
    } else setCart([...cart, { ...item, cartId: itemId }]);
    setSelectedProduct(null);
  };
  const updateCartQty = (id, delta) => setCart(cart.map(i => i.cartId === id ? { ...i, qty: i.qty + delta } : i).filter(i => i.qty > 0));
  const removeFromCart = (id) => setCart(cart.filter(i => i.cartId !== id));

  return (
    <StoreContext.Provider value={{
      db: { users, products, categories, settings, themes: THEMES, activeTheme },
      admin: { setUsers, setProducts, setSettings, setActiveTheme, currentUser, setCurrentUser },
      cartState: { cart, addToCart, updateCartQty, removeFromCart, total: cart.reduce((s, i) => s + (i.unitPrice * i.qty), 0), count: cart.reduce((s, i) => s + i.qty, 0), isCartOpen, setIsCartOpen },
      ui: { selectedProduct, setSelectedProduct, isSearchOpen, setIsSearchOpen }
    }}>
      {children}
    </StoreContext.Provider>
  );
};

const ProductModal = () => {
  const { ui, cartState } = useStore();
  const product = ui.selectedProduct;
  const [selectedVariant, setSelectedVariant] = useState(null);
  const [selectedAddons, setSelectedAddons] = useState([]);
  const [qty, setQty] = useState(1);
  const [notes, setNotes] = useState('');

  useEffect(() => {
    if (product) {
      setSelectedVariant(product.variants?.length > 0 ? product.variants[0] : null);
      setSelectedAddons([]); setQty(1); setNotes('');
    }
  }, [product]);

  if (!product) return null;

  const toggleAddon = (addon) => {
    const exists = selectedAddons.find(a => a.id === addon.id);
    if (exists) setSelectedAddons(selectedAddons.filter(a => a.id !== addon.id));
    else setSelectedAddons([...selectedAddons, addon]);
  };

  const unitPrice = (selectedVariant ? selectedVariant.price : (product.salePrice || product.basePrice)) + selectedAddons.reduce((s, a) => s + a.price, 0);

  return (
    <div className="fixed inset-0 z-50 flex items-end md:items-center justify-center p-0 md:p-6">
      <div className="absolute inset-0 bg-black/60 backdrop-blur-sm transition-opacity" onClick={() => ui.setSelectedProduct(null)}></div>
      
      <div className="theme-card w-full max-w-2xl md:rounded-[3rem] rounded-t-[3rem] overflow-hidden flex flex-col max-h-[90vh] md:max-h-[85vh] relative z-10 animate-slide-up shadow-2xl">
        <button onClick={() => ui.setSelectedProduct(null)} className="absolute top-6 right-6 z-20 bg-white/20 backdrop-blur-md p-3 rounded-full text-white shadow-lg hover:bg-white/30 transition">
          <X size={20} />
        </button>

        <div className="overflow-y-auto flex-grow pb-32 no-scrollbar">
          <div className="w-full h-72 md:h-96 relative">
             {product.image ? <img src={product.image} className="w-full h-full object-cover" /> : <LuxuryPlaceholder />}
             <div className="absolute inset-0 bg-gradient-to-t from-[var(--color-card-bg)] via-transparent to-transparent"></div>
          </div>
          
          <div className="px-6 md:px-10 pb-6 -mt-10 relative z-10">
            <h2 className="text-3xl md:text-4xl font-black theme-text mb-3">{product.name}</h2>
            <p className="text-lg theme-text-muted leading-relaxed mb-6">{product.desc}</p>
            
            <div className="space-y-8">
              {product.variants?.length > 0 && (
                <div>
                  <h3 className="font-bold theme-text mb-4 flex items-center gap-2"><div className="w-1.5 h-6 theme-primary-bg rounded-full"></div> اختر الحجم</h3>
                  <div className="grid gap-3">
                    {product.variants.map(v => (
                      <label key={v.id} onClick={() => setSelectedVariant(v)} className={`flex items-center justify-between p-5 rounded-2xl cursor-pointer transition-all ${selectedVariant?.id === v.id ? 'theme-primary-bg shadow-lg' : 'border border-black/10 hover:border-[var(--color-primary)]'}`}>
                        <span className={`font-bold ${selectedVariant?.id === v.id ? 'text-white' : 'theme-text'}`}>{v.name}</span>
                        <span className={`font-black ${selectedVariant?.id === v.id ? 'text-white' : 'theme-primary-text'}`}>{v.price} ﷼</span>
                      </label>
                    ))}
                  </div>
                </div>
              )}

              {product.addons?.length > 0 && (
                <div>
                  <h3 className="font-bold theme-text mb-4 flex items-center gap-2"><div className="w-1.5 h-6 theme-primary-bg rounded-full"></div> إضافات (اختياري)</h3>
                  <div className="grid gap-3">
                    {product.addons.map(a => {
                      const isSelected = selectedAddons.some(sa => sa.id === a.id);
                      return (
                        <label key={a.id} onClick={() => toggleAddon(a)} className={`flex items-center justify-between p-5 rounded-2xl cursor-pointer transition-all ${isSelected ? 'theme-primary-bg shadow-lg' : 'border border-black/10 hover:border-[var(--color-primary)]'}`}>
                          <div className="flex items-center gap-3">
                            <div className={`w-6 h-6 rounded-md border-2 flex items-center justify-center ${isSelected ? 'border-white' : 'border-gray-300'}`}>
                              {isSelected && <Check size={16} className="text-white" />}

                            <span className={`font-bold ${isSelected ? 'text-white' : 'theme-text'}`}>{a.name}</span>
                          </div>
                          <span className={`font-black ${isSelected ? 'text-white' : 'theme-text-muted'}`}>+{a.price} ﷼</span>
                        </label>
                      );
                    })}
                  </div>
                </div>
              )}

              <div>
                <h3 className="font-bold theme-text mb-4 flex items-center gap-2"><div className="w-1.5 h-6 theme-primary-bg rounded-full"></div> ملاحظات خاصة</h3>
                <textarea className="w-full bg-black/5 border-none rounded-2xl p-5 focus:ring-2 focus:ring-[var(--color-primary)] outline-none theme-text resize-none" rows="3" placeholder="بدون مكسرات..." value={notes} onChange={(e) => setNotes(e.target.value)}></textarea>

            </div>

                <span className="w-8 text-center font-black text-xl theme-text">{qty}</span>

             <button onClick={() => cartState.addToCart({ product, variant: selectedVariant, addons: selectedAddons, notes, qty, unitPrice })} className="flex-grow theme-primary-bg h-[68px] rounded-2xl font-bold text-lg flex items-center justify-between px-8 shadow-[0_10px_30px_rgba(var(--color-primary),0.3)] hover:scale-[1.02] transition-transform">

             </button>
           </div>
        </div>
      </div>
    </div>
  );
};

const CartDrawer = () => {

  const handleOrder = () => {
    let msg = `السلام عليكم 🌿\nأرغب في تجهيز طلب من ${db.settings.name}:\n\n`;
    cart.forEach((item, i) => {
      msg += `▪️ ${item.product.name} ${item.variant ? `(${item.variant.name})` : ''}\n`;
      if (item.addons.length) msg += `   إضافات: ${item.addons.map(a => a.name).join('، ')}\n`;
      if (item.notes) msg += `   ملاحظة: ${item.notes}\n`;
      msg += `   الكمية: ${item.qty} | السعر: ${(item.unitPrice * item.qty).toFixed(2)} ﷼\n\n`;

    window.open(`https://wa.me/${db.settings.whatsapp.replace(/\+/g, '')}?text=${encodeURIComponent(msg)}`, '_blank');
  };

  if (!isCartOpen) return null;

                   <div className="flex-grow">

const AdminLogin = () => {
  const { db, admin } = useStore();
  const [user, setUser] = useState('');
  const [pass, setPass] = useState('');
  const [error, setError] = useState('');

  const handleLogin = (e) => {
    e.preventDefault();
    const matchedUser = db.users.find(u => u.username.toLowerCase() === user.toLowerCase() && u.password.toLowerCase() === pass.toLowerCase());
    
    if (matchedUser) {
      if (matchedUser.active) {
        admin.setCurrentUser(matchedUser);
      } else {
        setError('تم إيقاف هذا الحساب من قبل الإدارة.');
      }
    } else {
      setError('بيانات الدخول غير صحيحة أو غير مصرح بها');
    }
  };

  return (
    <div className="min-h-screen bg-[#121212] flex items-center justify-center p-6" dir="rtl">
      <div className="bg-[#1E1E1E] p-10 rounded-[2rem] shadow-2xl w-full max-w-md border border-white/10 relative overflow-hidden">
        
        {/* Return to main site subtly */}
        <button onClick={() => window.location.hash = ''} className="absolute top-6 left-6 text-white/30 hover:text-white transition">
          <ArrowRight size={24} />
        </button>

        <div className="text-center mb-10 mt-4">
          <div className="w-20 h-20 bg-[#D4AF37]/10 text-[#D4AF37] rounded-3xl flex items-center justify-center mx-auto mb-6"><Lock size={40} /></div>
          <h1 className="text-3xl font-black text-white mb-2">النظام الأمني المغلق</h1>
          <p className="text-white/50">محاولة دخول مصرحة فقط للآدمن</p>
        </div>
        <form onSubmit={handleLogin} className="space-y-6">
          <input type="text" placeholder="معرف المستخدم" required value={user} onChange={e=>setUser(e.target.value)} className="w-full px-5 py-4 bg-black/30 rounded-xl border border-white/10 text-white focus:border-[#D4AF37] focus:ring-1 focus:ring-[#D4AF37] outline-none transition" dir="ltr" />
          <input type="password" placeholder="كلمة المرور الأمنية" required value={pass} onChange={e=>setPass(e.target.value)} className="w-full px-5 py-4 bg-black/30 rounded-xl border border-white/10 text-white focus:border-[#D4AF37] focus:ring-1 focus:ring-[#D4AF37] outline-none transition" dir="ltr" />
          {error && <div className="bg-red-500/10 border border-red-500/50 text-red-400 p-3 rounded-lg text-sm text-center font-bold">{error}</div>}
          <button type="submit" className="w-full bg-[#D4AF37] text-black py-4 rounded-xl font-black text-lg hover:bg-[#b5952f] transition shadow-[0_10px_30px_rgba(212,175,55,0.2)]">تحقق ودخول</button>
        </form>
      </div>
    </div>
  );
};

const AdminDashboard = () => {
  const { db, admin } = useStore();
  const [activeTab, setActiveTab] = useState('products');
  const isSuperAdmin = admin.currentUser.role === 'super_admin';

  const [editingProduct, setEditingProduct] = useState(null);
  const [isAddingProduct, setIsAddingProduct] = useState(false);
  const [formData, setFormData] = useState({});

  const openEditor = (prod = null) => {
    setFormData(prod || {
      id: `p_${Date.now()}`, name: '', desc: '', categoryId: db.categories[0].id, image: '', 
      basePrice: 0, salePrice: '', discountPercent: '', calories: '', 
      active: true, featured: false, bestSeller: false, variants: [], addons: []
    });
    setEditingProduct(prod ? true : false);
    setIsAddingProduct(true);
  };

  const handleImageUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onloadend = () => {
        setFormData({ ...formData, image: reader.result });
      };
      reader.readAsDataURL(file);
    }
  };

    if (editingProduct) admin.setProducts(db.products.map(p => p.id === formData.id ? formData : p));
    else admin.setProducts([...db.products, formData]);

        {activeTab === 'products' && !isAddingProduct && (

const AppRouter = () => {
  const [currentPath, setCurrentPath] = useState(window.location.hash);
  const { admin } = useStore();

  useEffect(() => {
    const handleHashChange = () => setCurrentPath(window.location.hash);
    window.addEventListener('hashchange', handleHashChange);
    return () => window.removeEventListener('hashchange', handleHashChange);
  }, []);

  const isAdminRoute = currentPath.startsWith('#/hzm-secure-vault-26');

  if (isAdminRoute) {
    if (!admin.currentUser) return <AdminLogin />;
    return <AdminDashboard />;
  }
  return <CustomerApp />;
};

export default function App() {
  return (
    <StoreProvider>
      <AppRouter />
    </StoreProvider>
  );
}
